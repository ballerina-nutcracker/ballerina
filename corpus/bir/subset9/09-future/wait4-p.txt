module $anon.. v 0.0.0;
crash() -> int{

    %1 = ConstantLoad alternate wait panic
    %2 = newError error(%1)
    panic %2;
  }
}
number() -> int{
  bb0 {
    %1 = ConstantLoad 1
    %0 = %1;
    return;
  }
}
main() -> nil{
  bb0 {
    %1 = start crash() isolated=false -> bb1;
  }
  bb1 {
    panicked = %1;
    %3 = start number() isolated=false -> bb2;
  }
  bb2 {
    succeeded = %3;
    %5 = wait panicked | succeeded -> bb3;
  }
  bb3 {
    result = %5;
    %7 = result is int
    %8 = %7;
    return;
  }
}
