module $anon.. v 0.0.0;
counter  int;
spawner() -> int{

    %1 = ConstantLoad before
    %2 = println(%1) -> bb1;
  }
  bb1 {
    %3 = createLatch() -> bb2;
  }

    %7 = closure_fp $anon/.:$worker:spawner:$function
    $desugar$3 = %7;
    %9 = closure_fp $anon/.:$worker:spawner:w
    $desugar$4 = %9;
    %11 = start $desugar$3() isolated=true -> bb3;
  }
  bb3 {
    $desugar$1 = %11;
    %12 = start $desugar$4() isolated=true -> bb4;

  bb4 {

  }

main() -> nil{

  }
  bb1 {
    PushScopeFrame 1
    %0 = spawner() -> bb2;
  }
  bb2 {
    counter = %0;
    (1, result) = counter;
    PopScopeFrame
    lock-end "$anon/.:counter" GOTO bb3;
  }
  bb3 {
    %2 = result;
    %3 = println(%2) -> bb4;
  }
  bb4 {
    return;
  }
}

  }
  bb3 {

}
$worker:spawner:w() -> int{
